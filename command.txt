#J'ai fait ca avec Maven 3.8.7

#Compiler
mvn compile / mvn clean compile

#Executer un fichier 
mvn exec:java -Dexec.mainClass="com.memoform.main.ViewMain"

#Lancer les tests
mvn test

#Créer le jar
mvn package

#Executer le jar
java -jar target/projet-memoform-1.0-SNAPSHOT.jar
